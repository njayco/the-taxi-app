Yep — that “pin jumps to the upper-left corner” bug is almost always one of these:
	1.	Marker is being recreated on re-render (React state change causes a new Marker without coordinates yet)
	2.	Coordinates temporarily become undefined/null during an update → Mapbox treats it like (0,0) (which appears upper-left-ish depending on your view)
	3.	Map container/layout shifts (sidebar height changes, fonts load, etc.) and Mapbox needs a map.resize()
	4.	Marker CSS positioning gets clobbered (less common, but happens if a global CSS hits .mapboxgl-marker)

Below is the fix pattern that reliably stops the jumping: create the Map once, keep Markers in refs, and only update marker position when coords are valid.

⸻

Fix: stop the pin from jumping

Core rules
	•	Initialize map one time (useEffect(() => {...}, []))
	•	Keep a markersByCallId ref and update markers, don’t recreate them
	•	Guard against invalid coords (Number.isFinite(lat/lng))
	•	When the left panel expands/collapses (or list changes), call map.resize()

Drop-in pattern (Mapbox GL + React)

// utils/coords.ts
export function isValidLngLat(lng: any, lat: any) {
  const L = Number(lng);
  const A = Number(lat);
  return Number.isFinite(L) && Number.isFinite(A) && Math.abs(L) <= 180 && Math.abs(A) <= 90;
}

// in your DispatchMap component
import mapboxgl from "mapbox-gl";
import { useEffect, useMemo, useRef } from "react";
import { isValidLngLat } from "@/utils/coords";

type CallPin = { id: string; lng: number; lat: number; status: "NEW" | "COMPLETED" | "PICKED_UP"; fare?: number };

export function DispatchMap({ calls }: { calls: CallPin[] }) {
  const mapRef = useRef<mapboxgl.Map | null>(null);
  const mapContainerRef = useRef<HTMLDivElement | null>(null);

  // Markers stored by call id so we UPDATE instead of recreate
  const callMarkersRef = useRef<Map<string, mapboxgl.Marker>>(new Map());

  // 1) Create map ONCE
  useEffect(() => {
    if (!mapContainerRef.current || mapRef.current) return;

    mapRef.current = new mapboxgl.Map({
      container: mapContainerRef.current,
      style: "mapbox://styles/mapbox/streets-v12",
      center: [-73.616, 40.706], // Hempstead-ish
      zoom: 12,
    });

    return () => {
      mapRef.current?.remove();
      mapRef.current = null;
      callMarkersRef.current.clear();
    };
  }, []);

  // 2) When calls change, update markers safely
  useEffect(() => {
    const map = mapRef.current;
    if (!map) return;

    const existing = callMarkersRef.current;

    // Add/update
    for (const c of calls) {
      if (!isValidLngLat(c.lng, c.lat)) continue; // prevents (0,0) jumps

      const key = c.id;
      const m = existing.get(key);

      if (!m) {
        const el = document.createElement("div");
        el.className = "call-marker";
        // optional: status-based styling via class
        el.dataset.status = c.status;

        const marker = new mapboxgl.Marker({ element: el, anchor: "bottom" })
          .setLngLat([c.lng, c.lat])
          .addTo(map);

        existing.set(key, marker);
      } else {
        // only setLngLat when valid
        m.setLngLat([c.lng, c.lat]);
        const el = m.getElement();
        el.dataset.status = c.status;
      }
    }

    // Remove markers for calls that no longer exist
    const ids = new Set(calls.map(c => c.id));
    for (const [id, marker] of existing.entries()) {
      if (!ids.has(id)) {
        marker.remove();
        existing.delete(id);
      }
    }
  }, [calls]);

  // 3) Resize map if layout changes (sidebar/filter toggles)
  useEffect(() => {
    const map = mapRef.current;
    if (!map) return;
    // run next tick so DOM has settled
    const t = window.setTimeout(() => map.resize(), 0);
    return () => window.clearTimeout(t);
  }, [calls.length]); // you can also depend on sidebarOpen, filters, etc.

  return <div ref={mapContainerRef} className="h-full w-full" />;
}

// /app/api/calls/route.ts
import { NextResponse } from "next/server";

function startOfDay(d: Date) {
  const x = new Date(d);
  x.setHours(0,0,0,0);
  return x;
}

function rangeToStart(range: string) {
  const now = new Date();
  const end = now;
  let start = new Date(0);

  if (range === "TODAY") start = startOfDay(now);
  else if (range === "LAST_7_DAYS") start = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
  else if (range === "LAST_30_DAYS") start = new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000);
  else if (range === "LAST_6_MONTHS") start = new Date(now.getTime() - 183 * 24 * 60 * 60 * 1000);
  else if (range === "LAST_12_MONTHS") start = new Date(now.getTime() - 365 * 24 * 60 * 60 * 1000);
  else if (range === "ALL_TIME") start = new Date(0);

  return { start, end };
}

export async function GET(req: Request) {
  const { searchParams } = new URL(req.url);
  const status = (searchParams.get("status") ?? "ALL").toUpperCase();
  const range = (searchParams.get("range") ?? "TODAY").toUpperCase();

  const { start, end } = rangeToStart(range);

  // TODO: replace with your real DB fetch
  const allCalls = globalThis.callsStore ?? []; // example
  const callsInRange = allCalls.filter((c: any) => {
    const created = new Date(c.createdAt);
    return created >= start && created <= end;
  });

  const callsFiltered =
    status === "ALL" ? callsInRange : callsInRange.filter((c: any) => c.status === status);

  const sumFare = (rows: any[]) =>
    rows.reduce((acc, c) => acc + (Number(c.fare) || 0), 0);

  const revenueByStatus = {
    NEW: sumFare(callsInRange.filter((c: any) => c.status === "NEW")),
    COMPLETED: sumFare(callsInRange.filter((c: any) => c.status === "COMPLETED" || c.status === "PICKED_UP")),
  };

  const revenueTotal =
    status === "NEW" ? revenueByStatus.NEW :
    status === "COMPLETED" ? revenueByStatus.COMPLETED :
    revenueByStatus.NEW + revenueByStatus.COMPLETED;

  return NextResponse.json({
    calls: callsFiltered,
    revenue: { total: revenueTotal, byStatus: revenueByStatus }
  });
}
