export function normalizeToMs(lastSeenAt: string | number | Date | null): number {
  if (!lastSeenAt) return 0;
  if (typeof lastSeenAt === "number") {
    // if seconds epoch, convert to ms
    return lastSeenAt < 10_000_000_000 ? lastSeenAt * 1000 : lastSeenAt;
  }
  return new Date(lastSeenAt).getTime();
}

export function computeDriverStatus(lastSeenAt: string | number | Date | null, nowMs: number) {
  const lastSeenMs = normalizeToMs(lastSeenAt);
  const ageMs = Math.max(0, nowMs - lastSeenMs);
  const ONLINE_THRESHOLD_MS = 15_000;
  return {
    ageMs,
    status: lastSeenMs > 0 && ageMs <= ONLINE_THRESHOLD_MS ? "ONLINE" : "OFFLINE",
  };
}
